do $$
declare v_def text; v_nuevo text; v_i int; v_j int;
  v_a constant text := E'  if v_pago.estado in (''APROBADO'',''RECHAZADO'',''REEMBOLSADO'',''REEMBOLSADO_PARCIAL'') then';
  v_ini constant text := E'  -- Barberos adicionales (migración 175)';
begin
  v_def := pg_get_functiondef('public.aplicar_evento_pago(uuid,text,pago_estado,jsonb)'::regprocedure);
  v_i := position(v_ini in v_def);
  if v_i > 0 then
    v_j := position(v_a in v_def);
    v_nuevo := substr(v_def, 1, v_i - 1) || substr(v_def, v_j);
    execute v_nuevo;
  end if;
end $$;

do $$
declare v_def text; v_f text;
  v_x constant text := ' and coalesce(metadata->>''concepto'', '''') <> ''STAFF_EXTRA''';

  end loop;
end $$;
