begin;
create or replace function public.negocio_puede_recibir_reservas(p_negocio_id uuid)
 returns boolean language sql stable security definer set search_path to 'public'
as $function$
  select coalesce((
    select s.estado in ('ACTIVA','TRIAL') or (s.estado = 'EN_MORA' and (s.gracia_fin is null or s.gracia_fin > now()))
    from public.suscripcion s where s.negocio_id = p_negocio_id
  ), true);
$function$;

do $$
declare v_def text; v_nuevo text;
begin
  v_def := pg_get_functiondef('public.trg_reserva_exige_plan()'::regprocedure);
  v_nuevo := regexp_replace(v_def, 'raise exception ''SUSCRIPCION_RESTRINGIDA''[^;]*;', 'raise exception ''NEGOCIO_REQUIERE_PLAN'';');
  if v_nuevo = v_def then raise exception 'ANCLA_NO_ENCONTRADA trg_reserva_exige_plan (reversión)'; end if;
  execute v_nuevo;
end $$;
